```ts
import type { BadgeId, Contact, Etape } from './types';

export interface ContactFilters {
  recherche: string;
  badges: BadgeId[];
  etape: Etape | 'toutes';
}

export function filterContacts(
  contacts: Contact[],
  filters: ContactFilters,
): Contact[] {
  const recherche = filters.recherche.trim().toLowerCase();
  return contacts.filter((contact) => {
    if (filters.etape !== 'toutes' && contact.etape !== filters.etape) {
      return false;
    }
    if (
      filters.badges.length > 0 &&
      !filters.badges.some((b) => contact.badges.includes(b))
    ) {
      return false;
    }
    if (recherche) {
      const champs = [
        contact.prenom,
        contact.nom,
        contact.organisation ?? '',
        contact.email ?? '',
      ]
        .join(' ')
        .toLowerCase();
      if (!champs.includes(recherche)) {
        return false;
      }
    }
    return true;
  });
}
```

```tsx
function initiales(contact: Contact): string {

  contact,
```

```tsx
function valeursInitiales(contact: Contact | null): ContactFormValues {
  return {
    prenom: contact?.prenom ?? '',
    nom: contact?.nom ?? '',
    organisation: contact?.organisation ?? '',
    fonction: contact?.fonction ?? '',
    email: contact?.email ?? '',
    telephone: contact?.telephone ?? '',
    etape: contact?.etape ?? 'a_contacter',
    badges: contact?.badges ?? [],
    notes: contact?.notes ?? '',
  };

export function ContactFiche({
  open,
  contact,
  onClose,
  onSave,
  onDelete,
}: {
  open: boolean;
  contact: Contact | null;
  onClose: () => void;
  onSave: (values: ContactFormValues) => void;
  onDelete: () => void;
}) {
  const [values, setValues] = useState<ContactFormValues>(
    valeursInitiales(contact),
  );

  useEffect(() => {
    setValues(valeursInitiales(contact));
  }, [contact]);

  function set<K extends keyof ContactFormValues>(
    key: K,
    value: ContactFormValues[K],
  ) {
    setValues((prev) => ({ ...prev, [key]: value }));

    setValues((prev) => ({

      badges: prev.badges.includes(badge)

      title={contact ? 'Modifier le contact' : 'Nouveau contact'}
      onClose={onClose}
```

```tsx
import { useCallback, useEffect, useMemo, useState } from 'react';
import { PageHeader } from '../../components/ui/PageHeader';
import { Button } from '../../components/ui/Button';
import { EmptyState } from '../../components/ui/EmptyState';
import { DemoDataToggle } from '../../components/DemoDataToggle';
import { contactsStore } from './store';
import { ContactCard } from './ContactCard';
import { ContactListView } from './ContactListView';
import { ContactFiche } from './ContactFiche';
import { CrmFilters } from './CrmFilters';
import { filterContacts, type ContactFilters } from './filter';
import type { Contact, ContactFormValues } from './types';

type Vue = 'cartes' | 'liste';

function aujourdhui(): string {
  return new Date().toISOString().slice(0, 10);
}

function vide(v: string): string | undefined {
  const t = v.trim();
  return t === '' ? undefined : t;
}

export default function CrmPage() {
  const [contacts, setContacts] = useState<Contact[]>([]);
  const [filtres, setFiltres] = useState<ContactFilters>({
    recherche: '',
    badges: [],
    etape: 'toutes',
  });
  const [vue, setVue] = useState<Vue>('cartes');
  const [ficheOuverte, setFicheOuverte] = useState(false);
  const [contactEdite, setContactEdite] = useState<Contact | null>(null);

  const rafraichir = useCallback(() => {
    void contactsStore.list().then(setContacts);
  }, []);

  useEffect(() => {
    rafraichir();
  }, [rafraichir]);

  const filtres_resultat = useMemo(
    () => filterContacts(contacts, filtres),
    [contacts, filtres],
  );

  function ouvrirCreation() {
    setContactEdite(null);
    setFicheOuverte(true);
  }

  function ouvrirEdition(contact: Contact) {
    setContactEdite(contact);
    setFicheOuverte(true);
  }

  async function enregistrer(values: ContactFormValues) {
    const champs = {
      prenom: values.prenom.trim(),
      nom: values.nom.trim(),
      organisation: vide(values.organisation),
      fonction: vide(values.fonction),
      email: vide(values.email),
      telephone: vide(values.telephone),
      etape: values.etape,
      badges: values.badges,
      notes: vide(values.notes),
    };
    if (contactEdite) {
      await contactsStore.update(contactEdite.id, {
        ...champs,
        dateDerniereInteraction: aujourdhui(),
      });
    } else {
      await contactsStore.create({
        ...champs,
        source: 'manuel',
        dateCreation: aujourdhui(),
        dateDerniereInteraction: aujourdhui(),
      });
    }
    setFicheOuverte(false);
    rafraichir();
  }

  async function supprimer() {
    if (contactEdite) {
      await contactsStore.remove(contactEdite.id);
    }
    setFicheOuverte(false);
    rafraichir();
  }

  const vide_total = contacts.length === 0;

  return (
    <>
      <PageHeader
        title="CRM"
        description="Contacts, prospection et badges."
        actions={
          <>
            <DemoDataToggle store={contactsStore} onChange={rafraichir} />
            <Button variant="primary" onClick={ouvrirCreation}>
              Nouveau contact
            </Button>
          </>
        }
      />

      <div className="mb-4 flex items-center gap-1.5">
        <Button
          variant={vue === 'cartes' ? 'primary' : 'ghost'}
          onClick={() => setVue('cartes')}
        >
          Cartes
        </Button>
        <Button
          variant={vue === 'liste' ? 'primary' : 'ghost'}
          onClick={() => setVue('liste')}
        >
          Liste
        </Button>
      </div>

      <CrmFilters value={filtres} onChange={setFiltres} />

      {vide_total ? (
        <EmptyState
          title="Aucun contact"
          description="Active les données de démo ou crée ton premier contact."
          action={
            <Button variant="primary" onClick={ouvrirCreation}>
              Nouveau contact
            </Button>
          }
        />
      ) : filtres_resultat.length === 0 ? (
        <EmptyState
          title="Aucun résultat"
          description="Aucun contact ne correspond aux filtres."
        />
      ) : vue === 'liste' ? (
        <ContactListView contacts={filtres_resultat} onSelect={ouvrirEdition} />
      ) : (
        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
          {filtres_resultat.map((contact) => (
            <ContactCard
              key={contact.id}
              contact={contact}
              onClick={() => ouvrirEdition(contact)}
            />
          ))}
        </div>
      )}

      <ContactFiche
        open={ficheOuverte}
        contact={contactEdite}
        onClose={() => setFicheOuverte(false)}
        onSave={enregistrer}
        onDelete={supprimer}
      />
    </>
  );
}
```
