# CRM vues supplémentaires (Plan 2b) : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ajouter quatre vues au CRM (Kanban par étape avec drag-and-drop, Groupé par badge, Relances à venir, Tableau de bord) et un sélecteur de vues unifié, en plus des vues Cartes et Liste existantes.

**Architecture:** Tout dans `web/src/features/crm/`. Chaque vue est un composant présentationnel autonome recevant `contacts` (et des callbacks) en props. La page `index.tsx` orchestre le choix de vue. Le Kanban modifie l'étape d'un contact via le store au drop. Le Tableau de bord affiche des stats globales (non filtrées) avec Recharts.

**Tech Stack:** React 19, TypeScript strict (`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`), Tailwind v4, Vitest + Testing Library, `@dnd-kit/core`, `recharts`.

**Référence:** sections 7.3 et 7.4 du spec. Vues existantes : Cartes (`ContactCard`), Liste (`ContactListView`).

**Décisions:** un contact multi-badges apparaît dans chaque section "Groupé par badge". La vue Relances ne montre que les contacts ayant `prochaineRelance`, triés par date, retards en `--erreur`. Le Tableau de bord ignore les filtres (stats globales) et masque la barre de filtres.

**Commandes de vérification:**

- Tests : `npm run test --workspace web`
- Typecheck : `npm run typecheck --workspace web`

```tsx
import {
  DndContext,
  PointerSensor,
  useDraggable,
  useDroppable,
  useSensor,
  useSensors,
  type DragEndEvent,
} from '@dnd-kit/core';
import { Avatar } from '../../components/ui/Avatar';
import { Badge } from '../../components/ui/Badge';
import { cn } from '../../lib/cn';
import {
  BADGE_LABELS,
  BADGE_TONE,
  ETAPES,
  ETAPE_LABELS,
  type Contact,
  type Etape,
} from './types';

function initiales(contact: Contact): string {
  return `${contact.prenom.charAt(0)}${contact.nom.charAt(0)}`.toUpperCase();
}

function KanbanCard({
  contact,
  onSelect,
}: {
  contact: Contact;
  onSelect: () => void;
}) {
  const { attributes, listeners, setNodeRef, transform, isDragging } =
    useDraggable({
      id: contact.id,
    });
  const style = transform
    ? { transform: `translate(${transform.x}px, ${transform.y}px)` }
    : undefined;
  return (
    <div
      ref={setNodeRef}
      style={style}
      {...listeners}
      {...attributes}
      onClick={onSelect}
      className={cn(
        'cursor-pointer rounded-xl border-2 border-projecteur bg-papier p-3',
        isDragging && 'opacity-50',
      )}
    >
      <div className="flex items-center gap-2">
        <Avatar initials={initiales(contact)} />
        <div className="min-w-0">
          <p className="truncate text-sm font-semibold text-encre">
            {contact.prenom} {contact.nom}
          </p>
          {contact.organisation && (
            <p className="truncate text-xs text-encre-3">
              {contact.organisation}
            </p>
          )}
        </div>
      </div>
      {contact.badges.length > 0 && (
        <div className="mt-2 flex flex-wrap gap-1">
          {contact.badges.map((b) => (
            <Badge key={b} tone={BADGE_TONE[b]}>
              {BADGE_LABELS[b]}
            </Badge>
          ))}
        </div>
      )}
    </div>
  );
}

function KanbanColumn({
  etape,
  contacts,
  onSelect,
}: {
  etape: Etape;
  contacts: Contact[];
  onSelect: (contact: Contact) => void;
}) {
  const { setNodeRef, isOver } = useDroppable({ id: etape });
  return (
    <div
      ref={setNodeRef}
      className={cn(
        'flex w-72 flex-shrink-0 flex-col gap-2 rounded-xl bg-scene p-2',
        isOver && 'ring-2 ring-gel-rose',
      )}
    >
      <div className="flex items-center justify-between px-1 py-1 text-sm font-semibold text-encre">
        <span>{ETAPE_LABELS[etape]}</span>
        <span className="rounded-full bg-papier px-2 text-xs text-encre-3">
          {contacts.length}
        </span>
      </div>
      {contacts.map((contact) => (
        <KanbanCard
          key={contact.id}
          contact={contact}
          onSelect={() => onSelect(contact)}
        />
      ))}
    </div>
  );
}

export function KanbanView({
  contacts,
  onChangeEtape,
  onSelect,
}: {
  contacts: Contact[];
  onChangeEtape: (id: string, etape: Etape) => void;
  onSelect: (contact: Contact) => void;
}) {
  const sensors = useSensors(
    useSensor(PointerSensor, { activationConstraint: { distance: 5 } }),
  );

  function handleDragEnd(event: DragEndEvent) {
    const { active, over } = event;
    if (!over) return;
    onChangeEtape(String(active.id), over.id as Etape);
  }

  return (
    <DndContext sensors={sensors} onDragEnd={handleDragEnd}>
      <div className="flex gap-3 overflow-x-auto pb-2">
        {ETAPES.map((etape) => (
          <KanbanColumn
            key={etape}
            etape={etape}
            contacts={contacts.filter((c) => c.etape === etape)}
            onSelect={onSelect}
          />
        ))}
      </div>
    </DndContext>
  );
}
```

```tsx
import { ContactCard } from './ContactCard';
import { BADGES, BADGE_LABELS, type Contact } from './types';

export function GroupedByBadgeView({
  contacts,
  onSelect,
}: {
  contacts: Contact[];
  onSelect: (contact: Contact) => void;
}) {
  const groupes = BADGES.map((badge) => ({
    badge,
    liste: contacts.filter((c) => c.badges.includes(badge)),
  })).filter((g) => g.liste.length > 0);

  return (
    <div className="flex flex-col gap-6">
      {groupes.map(({ badge, liste }) => (
        <section key={badge}>
          <h2 className="mb-3 flex items-center gap-2 text-sm font-bold text-encre">
            {BADGE_LABELS[badge]}
            <span className="rounded-full bg-scene px-2 text-xs font-medium text-encre-3">
              {liste.length}
            </span>
          </h2>
          <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
            {liste.map((contact) => (
              <ContactCard
                key={contact.id}
                contact={contact}
                onClick={() => onSelect(contact)}
              />
            ))}
          </div>
        </section>
      ))}
    </div>
  );
}
```

```tsx
import { EmptyState } from '../../components/ui/EmptyState';
import { cn } from '../../lib/cn';
import type { Contact } from './types';

function formatDate(iso: string): string {
  return new Date(iso).toLocaleDateString('fr-FR');
}

export function RelancesView({
  contacts,
  onSelect,
}: {
  contacts: Contact[];
  onSelect: (contact: Contact) => void;
}) {
  const aujourdhui = new Date().toISOString().slice(0, 10);
  const avecRelance = contacts
    .filter((c) => c.prochaineRelance)
    .sort((a, b) => (a.prochaineRelance! < b.prochaineRelance! ? -1 : 1));

  if (avecRelance.length === 0) {
    return (
      <EmptyState
        title="Aucune relance planifiée"
        description="Renseigne une date de prochaine relance sur une fiche contact."
      />
    );
  }

  return (
    <div className="flex flex-col gap-2">
      {avecRelance.map((contact) => {
        const enRetard = contact.prochaineRelance! < aujourdhui;
        return (
          <button
            key={contact.id}
            type="button"
            onClick={() => onSelect(contact)}
            className="flex items-center justify-between rounded-xl border border-trait bg-papier px-4 py-3 text-left transition-colors hover:bg-scene"
          >
            <span className="font-medium text-encre">
              {contact.prenom} {contact.nom}
              {contact.organisation && (
                <span className="ml-2 text-encre-3">
                  {contact.organisation}
                </span>
              )}
            </span>
            <span
              className={cn(
                'text-sm font-semibold tabular-nums',
                enRetard ? 'text-erreur' : 'text-encre-2',
              )}
            >
              {enRetard ? 'En retard - ' : ''}
              {formatDate(contact.prochaineRelance!)}
            </span>
          </button>
        );
      })}
    </div>
  );
}
```

Remplacer tout le contenu de `web/src/features/crm/index.tsx` par :

```tsx
import { useCallback, useEffect, useMemo, useState } from 'react';
import { PageHeader } from '../../components/ui/PageHeader';
import { Button } from '../../components/ui/Button';
import { EmptyState } from '../../components/ui/EmptyState';
import { DemoDataToggle } from '../../components/DemoDataToggle';
import { contactsStore } from './store';
import { ContactCard } from './ContactCard';
import { ContactListView } from './ContactListView';
import { ContactFiche } from './ContactFiche';
import { CrmFilters } from './CrmFilters';
import { CrmViewSwitcher } from './CrmViewSwitcher';
import { KanbanView } from './KanbanView';
import { GroupedByBadgeView } from './GroupedByBadgeView';
import { RelancesView } from './RelancesView';
import { CrmDashboardView } from './CrmDashboardView';
import { filterContacts, type ContactFilters } from './filter';
import type { Vue } from './views';
import type { Contact, ContactFormValues, Etape } from './types';

function aujourdhui(): string {
  return new Date().toISOString().slice(0, 10);
}

function vide(v: string): string | undefined {
  const t = v.trim();
  return t === '' ? undefined : t;
}

export default function CrmPage() {
  const [contacts, setContacts] = useState<Contact[]>([]);
  const [filtres, setFiltres] = useState<ContactFilters>({
    recherche: '',
    badges: [],
    etape: 'toutes',
  });
  const [vue, setVue] = useState<Vue>('cartes');
  const [ficheOuverte, setFicheOuverte] = useState(false);
  const [contactEdite, setContactEdite] = useState<Contact | null>(null);

  const rafraichir = useCallback(() => {
    void contactsStore.list().then(setContacts);
  }, []);

  useEffect(() => {
    rafraichir();
  }, [rafraichir]);

  const resultatsFiltres = useMemo(
    () => filterContacts(contacts, filtres),
    [contacts, filtres],
  );

  function ouvrirCreation() {
    setContactEdite(null);
    setFicheOuverte(true);
  }

  function ouvrirEdition(contact: Contact) {
    setContactEdite(contact);
    setFicheOuverte(true);
  }

  async function changerEtape(id: string, etape: Etape) {
    await contactsStore.update(id, { etape });
    rafraichir();
  }

  async function enregistrer(values: ContactFormValues) {
    const requis = {
      prenom: values.prenom.trim(),
      nom: values.nom.trim(),
      etape: values.etape,
      badges: values.badges,
    };
    const opts: Partial<
      Pick<
        Contact,
        'organisation' | 'fonction' | 'email' | 'telephone' | 'notes'
      >
    > = {};
    const org = vide(values.organisation);
    if (org !== undefined) opts.organisation = org;
    const fn = vide(values.fonction);
    if (fn !== undefined) opts.fonction = fn;
    const em = vide(values.email);
    if (em !== undefined) opts.email = em;
    const tel = vide(values.telephone);
    if (tel !== undefined) opts.telephone = tel;
    const nt = vide(values.notes);
    if (nt !== undefined) opts.notes = nt;

    if (contactEdite) {
      await contactsStore.update(contactEdite.id, {
        ...requis,
        ...opts,
        dateDerniereInteraction: aujourdhui(),
      });
    } else {
      await contactsStore.create({
        ...requis,
        ...opts,
        source: 'manuel',
        dateCreation: aujourdhui(),
        dateDerniereInteraction: aujourdhui(),
      });
    }
    setFicheOuverte(false);
    rafraichir();
  }

  async function supprimer() {
    if (contactEdite) {
      await contactsStore.remove(contactEdite.id);
    }
    setFicheOuverte(false);
    rafraichir();
  }

  const aucunContact = contacts.length === 0;
  const tableau = vue === 'tableau';

  function rendreVue() {
    if (vue === 'tableau') {
      return <CrmDashboardView contacts={contacts} />;
    }
    if (resultatsFiltres.length === 0) {
      return (
        <EmptyState
          title="Aucun résultat"
          description="Aucun contact ne correspond aux filtres."
        />
      );
    }
    if (vue === 'liste') {
      return (
        <ContactListView contacts={resultatsFiltres} onSelect={ouvrirEdition} />
      );
    }
    if (vue === 'kanban') {
      return (
        <KanbanView
          contacts={resultatsFiltres}
          onChangeEtape={changerEtape}
          onSelect={ouvrirEdition}
        />
      );
    }
    if (vue === 'badges') {
      return (
        <GroupedByBadgeView
          contacts={resultatsFiltres}
          onSelect={ouvrirEdition}
        />
      );
    }
    if (vue === 'relances') {
      return (
        <RelancesView contacts={resultatsFiltres} onSelect={ouvrirEdition} />
      );
    }
    return (
      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4">
        {resultatsFiltres.map((contact) => (
          <ContactCard
            key={contact.id}
            contact={contact}
            onClick={() => ouvrirEdition(contact)}
          />
        ))}
      </div>
    );
  }

  return (
    <>
      <PageHeader
        title="CRM"
        description="Contacts, prospection et badges."
        actions={
          <>
            <DemoDataToggle store={contactsStore} onChange={rafraichir} />
            <Button variant="primary" onClick={ouvrirCreation}>
              Nouveau contact
            </Button>
          </>
        }
      />

      <div className="mb-4">
        <CrmViewSwitcher value={vue} onChange={setVue} />
      </div>

      {!tableau && <CrmFilters value={filtres} onChange={setFiltres} />}

      {aucunContact ? (
        <EmptyState
          title="Aucun contact"
          description="Active les données de démo ou crée ton premier contact."
          action={
            <Button variant="primary" onClick={ouvrirCreation}>
              Nouveau contact
            </Button>
          }
        />
      ) : (
        rendreVue()
      )}

      <ContactFiche
        open={ficheOuverte}
        contact={contactEdite}
        onClose={() => setFicheOuverte(false)}
        onSave={enregistrer}
        onDelete={supprimer}
      />
    </>
  );
}
```
