# Accueil (tableau de bord de synthèse) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development. Steps use checkbox (`- [ ]`) syntax.

**Goal:** Page d'accueil synthétisant l'activité (hors comptabilité) : prochains événements, tâches actives par membre, derniers contacts/leads, KPIs non financiers et raccourcis vers les onglets. Lecture seule des stores existants + un bouton global de données de démo.

**Architecture:** Module frontend `web/src/features/home/`. Fonctions pures de synthèse dans `summary.ts` (testées), page `index.tsx` qui lit `contactsStore`, `evenementsStore`, `tachesStore` et navigue via `Link` (react-router v7). Le bouton démo (de)seed les 5 stores de l'app.

**Tech Stack:** React 19, TypeScript strict (`noUncheckedIndexedAccess`, `exactOptionalPropertyTypes`), Tailwind v4, react-router v7, Vitest + Testing Library.

## Conventions verrouillées

- Français avec accents. Pas de tiret cadratin (—).
- Lecture seule : l'Accueil ne crée/modifie aucune donnée métier (sauf le bouton démo global qui appelle `seed()`/`clear()`).
- Le jaune (#F5DF4D / projecteur) ne porte jamais de texte (bordures/fills uniquement).
- Date du jour : `2026-06-14`.

Stores et types existants :

- CRM : `contactsStore` (`../crm/store`), `Contact`, `BADGE_LABELS`, `BADGE_TONE` (`../crm/types`).
- Calendrier : `evenementsStore`, `tachesStore` (`../calendar/store`), `Evenement`, `Tache`, `Membre`, `MEMBRES`, `STATUT_LABELS` (`../calendar/types`).
- Comptabilité : `mouvementsStore` (`../accounting/store`). Mail : `campagnesStore` (`../mail/store`). (Utilisés seulement par le bouton démo global.)

```ts
import type { Contact } from '../crm/types';
import {
  MEMBRES,
  type Evenement,
  type Membre,
  type Tache,
} from '../calendar/types';

export function evenementsAVenir(
  evenements: Evenement[],
  aujourdhui: string,
  limite = 5,
): Evenement[] {
  return [...evenements]
    .filter((e) => e.date >= aujourdhui)
    .sort((a, b) => (a.date < b.date ? -1 : a.date > b.date ? 1 : 0))
    .slice(0, limite);
}

export function tachesActives(taches: Tache[]): Tache[] {
  return taches.filter((t) => t.statut !== 'fait');
}

export interface ChargeMembre {
  membre: Membre;
  total: number;
}

// Nombre de tâches actives (non faites) par membre, dans l'ordre de MEMBRES.
export function chargeParMembre(taches: Tache[]): ChargeMembre[] {
  const actives = tachesActives(taches);
  return MEMBRES.map((membre) => ({
    membre,
    total: actives.filter((t) => t.membre === membre).length,
  }));
}

export function derniersContacts(contacts: Contact[], limite = 5): Contact[] {
  return [...contacts]
    .sort((a, b) =>
      a.dateCreation < b.dateCreation
        ? 1
        : a.dateCreation > b.dateCreation
          ? -1
          : 0,
    )
    .slice(0, limite);
}

export function compterLeads(contacts: Contact[]): number {
  return contacts.filter((c) => c.badges.includes('lead')).length;
}
```

```tsx
import HomePage from './index';

      <HomePage />
    </MemoryRouter>,

describe('HomePage', () => {
  beforeEach(() => localStorage.clear());

  it('affiche le titre et les raccourcis', () => {
    renderHome();

      screen.getByRole('heading', { name: 'Accueil' }),
    ).toBeInTheDocument();
    expect(screen.getByRole('link', { name: /CRM/ })).toBeInTheDocument();
    expect(
      screen.getByRole('link', { name: /Calendrier/ }),
    ).toBeInTheDocument();

    renderHome();
    fireEvent.click(screen.getByRole('button', { name: /Données de démo/ }));
    await waitFor(async () => {
      expect(await contactsStore.list()).toHaveLength(20);
      expect(await evenementsStore.list()).toHaveLength(20);
      expect(await tachesStore.list()).toHaveLength(20);
    });
  });
```

```tsx
import { useCallback, useEffect, useState } from 'react';
import { Link } from 'react-router';
import {
  ArrowRight,
  Calendar as CalendarIcon,
  Mail as MailIcon,
  Users,
  Wallet,
} from 'lucide-react';
import { PageHeader } from '../../components/ui/PageHeader';
import { Badge } from '../../components/ui/Badge';
import { Avatar } from '../../components/ui/Avatar';
import { cn } from '../../lib/cn';
import { contactsStore } from '../crm/store';
import { evenementsStore, tachesStore } from '../calendar/store';
import { mouvementsStore } from '../accounting/store';
import { campagnesStore } from '../mail/store';
import { BADGE_LABELS, BADGE_TONE, type Contact } from '../crm/types';
import type { Evenement, Tache } from '../calendar/types';
import {
  chargeParMembre,
  compterLeads,
  derniersContacts,
  evenementsAVenir,
  tachesActives,
} from './summary';

const AUJOURDHUI = '2026-06-14';

const TOUS_STORES = [
  contactsStore,
  evenementsStore,
  tachesStore,
  mouvementsStore,
  campagnesStore,
];

function formatDateCourte(iso: string): string {
  const [a, m, j] = iso.split('-');
  return `${j}/${m}/${a}`;
}

function KpiCard({ label, valeur }: { label: string; valeur: number }) {
  return (
    <div className="rounded-xl border-2 border-projecteur bg-papier p-4">
      <p className="text-xs font-medium uppercase tracking-wide text-encre-3">
        {label}
      </p>
      <p className="mt-1 text-3xl font-extrabold tabular-nums text-encre">
        {valeur}
      </p>
    </div>
  );
}

const RACCOURCIS = [
  { to: '/crm', label: 'CRM', icon: Users, desc: 'Contacts et prospection' },
  {
    to: '/calendrier',
    label: 'Calendrier',
    icon: CalendarIcon,
    desc: 'Événements et tâches',
  },
  {
    to: '/mail',
    label: 'Mail marketing',
    icon: MailIcon,
    desc: 'Leads et campagnes',
  },
  {
    to: '/comptabilite',
    label: 'Comptabilité',
    icon: Wallet,
    desc: 'Revenus et prévisionnel',
  },
];

function SectionCard({
  titre,
  lien,
  lienLabel,
  children,
}: {
  titre: string;
  lien: string;
  lienLabel: string;
  children: React.ReactNode;
}) {
  return (
    <section className="rounded-xl border-2 border-projecteur bg-papier p-4">
      <div className="mb-3 flex items-center justify-between">
        <h2 className="text-sm font-bold text-encre">{titre}</h2>
        <Link
          to={lien}
          className="inline-flex items-center gap-1 text-xs font-medium text-velours hover:underline"
        >
          {lienLabel}
          <ArrowRight size={14} />
        </Link>
      </div>
      {children}
    </section>
  );
}

export default function HomePage() {
  const [contacts, setContacts] = useState<Contact[]>([]);
  const [evenements, setEvenements] = useState<Evenement[]>([]);
  const [taches, setTaches] = useState<Tache[]>([]);
  const [demo, setDemo] = useState(false);

  const rafraichir = useCallback(() => {
    void contactsStore.list().then(setContacts);
    void evenementsStore.list().then(setEvenements);
    void tachesStore.list().then(setTaches);
    void contactsStore.isSeeded().then(setDemo);
  }, []);

  useEffect(() => {
    rafraichir();
  }, [rafraichir]);

  async function basculerDemo() {
    if (demo) {
      await Promise.all(TOUS_STORES.map((s) => s.clear()));
    } else {
      await Promise.all(TOUS_STORES.map((s) => s.seed()));
    }
    rafraichir();
  }

  const prochains = evenementsAVenir(evenements, AUJOURDHUI);
  const charges = chargeParMembre(taches);
  const recents = derniersContacts(contacts);
  const nbTachesActives = tachesActives(taches).length;
  const nbLeads = compterLeads(contacts);

  return (
    <>
      <PageHeader
        title="Accueil"
        description="Vue d'ensemble de l'activité (hors comptabilité)."
        actions={
          <button
            type="button"
            onClick={() => void basculerDemo()}
            className={cn(
              'inline-flex h-9 items-center gap-2 rounded-full border px-3 text-sm font-medium transition-colors',
              demo
                ? 'border-gel-rose bg-rose-brume text-velours'
                : 'border-trait-fort bg-papier text-encre-2',
            )}
          >
            <span
              className={cn(
                'h-2 w-2 rounded-full',
                demo ? 'bg-gel-rose' : 'bg-encre-4',
              )}
            />
            Données de démo {demo ? 'activées' : 'désactivées'}
          </button>
        }
      />

      <div className="mb-6 grid grid-cols-2 gap-3 lg:grid-cols-4">
        <KpiCard label="Contacts" valeur={contacts.length} />
        <KpiCard label="Leads" valeur={nbLeads} />
        <KpiCard label="Événements à venir" valeur={prochains.length} />
        <KpiCard label="Tâches à faire" valeur={nbTachesActives} />
      </div>

      <div className="mb-6 grid grid-cols-1 gap-4 lg:grid-cols-3">
        <SectionCard
          titre="Prochains événements"
          lien="/calendrier"
          lienLabel="Calendrier"
        >
          {prochains.length === 0 ? (
            <p className="text-sm text-encre-3">Aucun événement à venir.</p>
          ) : (
            <ul className="flex flex-col gap-2">
              {prochains.map((e) => (
                <li
                  key={e.id}
                  className="flex items-center justify-between gap-3 text-sm"
                >
                  <span className="text-encre">{e.titre}</span>
                  <span className="shrink-0 text-xs text-encre-3">
                    {formatDateCourte(e.date)}
                  </span>
                </li>
              ))}
            </ul>
          )}
        </SectionCard>

        <SectionCard
          titre="Tâches actives par membre"
          lien="/calendrier"
          lienLabel="Calendrier"
        >
          <ul className="flex flex-col gap-2">
            {charges.map(({ membre, total }) => (
              <li
                key={membre}
                className="flex items-center justify-between gap-3 text-sm"
              >
                <span className="text-encre">{membre}</span>
                <span className="rounded-full bg-scene px-2 text-xs font-medium tabular-nums text-encre-2">
                  {total}
                </span>
              </li>
            ))}
          </ul>
        </SectionCard>

        <SectionCard titre="Derniers contacts" lien="/crm" lienLabel="CRM">
          {recents.length === 0 ? (

            <ul className="flex flex-col gap-2">
              {recents.map((c) => (
                <li key={c.id} className="flex items-center gap-2.5">
                  <Avatar
                    initials={`${c.prenom[0] ?? ''}${c.nom[0] ?? ''}`.toUpperCase()}

                    <span className="block truncate text-sm font-medium text-encre">

                    </span>
                    {c.organisation && (

                      {BADGE_LABELS[c.badges[0]]}
                    </Badge>
                  )}
                </li>
              ))}

          )}
        </SectionCard>
      </div>

      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
```
